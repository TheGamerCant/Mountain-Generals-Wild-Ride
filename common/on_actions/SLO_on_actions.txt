on_actions = {
	on_startup = {
		effect = {
			SLO={
				remove_ideas = {
					SLO_DUMMY_economy_idea
					SLO_DUMMY_army_idea
				}
				country_event = { id = slovakia.85 }
			}
		}	
	}
	
	#Poland AI is kinda retarded, create a bunch of divisions to help them out
	on_declare_war = {
		effect = {

			if = {
				limit = {
					FROM = { original_tag = POL }
					tag = GER
					
					SLO = {
						is_in_faction_with = POL
						has_government = democratic
					}
				}
				
				POL = {
					hidden_effect={
						division_template = {
							name = "National Guard"
							division_names_group = POL_INF_01
							regiments = {
								infantry = { x = 0 y = 0 }
								infantry = { x = 0 y = 1 }
								infantry = { x = 0 y = 2 }
								infantry = { x = 1 y = 0 }
								infantry = { x = 1 y = 1 }
								infantry = { x = 1 y = 2 }
								infantry = { x = 2 y = 0 }
							}
							support = {
								engineer = { x = 0 y = 0 }
								artillery = { x = 0 y = 1 }
							}
						}
						
						for_loop_effect = {
							end = 32
							random_owned_state = {
								limit = { POL = { has_full_control_of_state = PREV } }
								create_unit = {
									division = "name = \"Gwardia Narodowa\" division_template = \"National Guard\" start_experience_factor = 0.1" 
									owner = POL
								}
							}
						}
					}
				}
			}
		}
	}
	
	on_weekly_SLO = {
		#No capitulating Czechia!!!1`1!!!!!
		effect = {
			if = {
				limit = { 
					NOT = { has_completed_focus = SLO_peace_with_the_czechs } 
					has_war_with = CZE
					OR = {
						controls_province = 583
						controls_province = 494
						controls_province = 445
						controls_province = 11556
						controls_province = 9569
						controls_province = 9541
						controls_province = 9421
						controls_province = 9429
						controls_province = 3462
						controls_province = 6418
					}
				}
				
				unlock_national_focus = SLO_peace_with_the_czechs
				CZE = { country_event = { id = slovakia.1 hours = 8 random_hours = 8 } }
			}
		}
	}
	
	#Random advisor events
	on_weekly_SLO = {
		effect = {
			if = {
				limit = { 
					has_power_balance = { id = SLO_encroaching_german_influence_bop }
					NOT = { has_country_flag = SLO_germany_appointed_a_new_advisor_flag }
					date > 1939.3.1
				}
				
				#2% chance by default
				random_list = {
					10 = {
						#0%
						modifier = {
							factor = 0
							power_balance_value = {
								id = SLO_encroaching_german_influence_bop
								value > 0.60
							}
						}
						#0.81%
						modifier = {
							factor = 0.4
							power_balance_value = {
								id = SLO_encroaching_german_influence_bop
								value > 0.40
							}
						}
						
						#2% -0.15 <-> 0.4
						
						#3.92%
						modifier = {
							factor = 2.0
							power_balance_value = {
								id = SLO_encroaching_german_influence_bop
								value < -0.15
							}
						}
						#5.77%
						modifier = {
							factor = 1.5
							power_balance_value = {
								id = SLO_encroaching_german_influence_bop
								value < -0.35
							}
						}
						#8.41%
						modifier = {
							factor = 1.5
							power_balance_value = {
								id = SLO_encroaching_german_influence_bop
								value < -0.70
							}
						}
						
						SLO_fire_event_to_pick_advisor = yes
					}
					
					490 = {}
				}
			}
		}
	}
	
	#If we are generating a war goal while going despotist, Germany will protest
	
	#Fires when we begin the manual justification
	on_generate_wargoal  = {
		effect = {
			if = {
				limit = {
					ROOT = {
						original_tag = SLO
						has_power_balance = { id = SLO_encroaching_german_influence_bop }
					}

					NOT = { AND = { 
						FROM = { original_tag = HUN }
						ROOT = { has_completed_focus = SLO_deal_with_the_magyar_once_and_for_all } 
					}}
				}
				
				add_to_array = { SLO.countries_generating_against = FROM }
				SLO = { country_event = { id = slovakia.84 days = 4 random_days = 4 } }
			}
		}
	}
}
